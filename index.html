Supabase Schema (PostgreSQL)

-- students
create table students (
  id text primary key, -- MDM001
  name text not null,
  class text, gender text,
  parent_name text, phone text,
  fees_status text default 'Due',
  created_at timestamp default now()
);
-- attendance
create table attendance (
  id uuid primary key default gen_random_uuid(),
  student_id text references students(id),
  date date, status text, -- Present/Absent/Late
  class text
);
-- fees
create table fees (
  id uuid primary key default gen_random_uuid(),
  student_id text references students(id),
  amount int, paid int, balance int,
  term text, status text
);
-- + staff, classes, subjects, exams,
--   expenses, notices, profiles, roles
-- RLS enabled on all tables

Next.js 14 Code

// lib/supabase.ts
import { createClient } from '@supabase/supabase-js'
export const supabase = createClient(
  process.env.NEXT_PUBLIC_SUPABASE_URL!,
  process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!
)

// app/students/page.tsx
export default async function Students(){
  const { data } = await supabase
    .from('students').select('*')
  return <StudentTable data={data} />
}

// Real-time attendance
supabase.channel('attendance')
 .on('postgres_changes',
  { event:'*', schema:'public', table:'attendance'},
  payload => console.log(payload)
 ).subscribe()
